function rateLimit({ name, windowSec, max, message }) {
  return async (req, res, next) => {
    if (!config.rateLimitEnabled) return next();
    try {
      const windowMs = windowSec * 1000;
      const bucket = Math.floor(Date.now() / windowMs);
      const filter = { key: `${name}:${req.ip}:${bucket}` };
      const update = {
        $inc: { count: 1 },
        $setOnInsert: { expiresAt: new Date((bucket + 2) * windowMs) },
      };
      let doc;
      try {
        doc = await RateLimit.findOneAndUpdate(filter, update, { upsert: true, new: true }).lean();
      } catch (e) {
        if (e.code !== 11000) throw e;
        // Hai request cÃ¹ng lÃºc cÃ¹ng táº¡o báº£n ghi Ä‘áº§u tiÃªn: thá»­ láº¡i má»™t láº§n, láº§n nÃ y báº£n ghi Ä‘Ã£ tá»“n táº¡i
        doc = await RateLimit.findOneAndUpdate(filter, update, { upsert: true, new: true }).lean();
      }
      if (doc.count > max) {
        const retry = Math.ceil(((bucket + 1) * windowMs - Date.now()) / 1000);
        res.setHeader('Retry-After', String(retry));
        return next(new HttpError(429, message || `Thao tÃ¡c quÃ¡ nhanh, hÃ£y thá»­ láº¡i sau ${retry} giÃ¢y`));

function buildAudioUrl(word) {
  if (typeof word !== 'string') return null;
  const slug = word.trim().toLowerCase().replace(/\s+/g, '_');
  if (!/^[a-z_-]+$/.test(slug)) return null;
  const first = slug[0];
  const three = slug.slice(0, 3).padEnd(3, '_');
  return `${audioBaseUrl}/${first}/${three}/${slug}/${slug}.mp3`;
}

function effectiveStreak(user) {
  const last = user.lastActiveDate;
  const alive = last === vnDate() || last === vnYesterday();
  return {
    currentStreak: alive ? user.currentStreak : 0,
    longestStreak: user.longestStreak,
  };

async function touchStreak(userId) {
  const today = vnDate();
  for (let i = 0; i < 3; i += 1) {
    const user = await User.findById(userId).select(
      'lastActiveDate currentStreak longestStreak'
    );
    if (!user) return null;
    if (user.lastActiveDate === today) {
      return { ...effectiveStreak(user), alreadyPinged: true };
    }

    const current =
      user.lastActiveDate === vnYesterday() ? user.currentStreak + 1 : 1;
    const longest = Math.max(user.longestStreak, current);

    const res = await User.updateOne(
      { _id: userId, lastActiveDate: user.lastActiveDate },
      {
        $set: {
          lastActiveDate: today,
          currentStreak: current,
          longestStreak: longest,
        },
        $push: { activeDays: { $each: [today], $slice: -400 } },
      }
    );
    if (res.modifiedCount === 1) {
      return { currentStreak: current, longestStreak: longest, alreadyPinged: false };
    }
  }
  const user = await User.findById(userId).select(
    'lastActiveDate currentStreak longestStreak'
  );
  return user ? { ...effectiveStreak(user), alreadyPinged: true } : null;
